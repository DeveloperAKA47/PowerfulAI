import io
import csv

@app.get("/books/export")

    def generate_csv():
        # 创建一个字符串缓冲区
        buffer = io.StringIO()
        writer = csv.DictWriter(buffer, fieldnames=["id", "title", "author"])
        writer.writeheader()
        yield buffer.getvalue()  # 先输出表头
        buffer.seek(0)
        buffer.truncate(0)
        
        for row in books:
            writer.writerow(row)
            yield buffer.getvalue()  # 逐行输出
            buffer.seek(0)
            buffer.truncate(0)
    
    return StreamingResponse(generate_csv(), media_type="text/csv", headers={"Content-Disposition": "attachment; filename=books.csv"})
